# 基于提示词优化的细粒度猫狗分类系统 - 毕业设计方案

## 1. 研究背景与目标

### 1.1 问题描述
- 细粒度图像分类：区分猫狗的不同品种（如波斯猫 vs 英短、柯基 vs 柴犬）
- 传统方法需要大量标注数据
- 基于 CLIP 的提示学习方法可以大幅降低数据需求

### 1.2 研究目标
1. **构建细粒度分类框架**：结合图像特征与文本语义的提示词优化方法
2. **设计动态提示词算法**：实现提示词的自适应调整
3. **开发可视化界面**：展示分类效果和模型决策过程

### 2.2 核心算法设计

#### 2.2.1 基础 CoOp 模型
```python
class CoOp:
    """
    Context Optimization for CLIP
    将可学习的 tokens 加入到文本提示词中
    """
    def __init__(self, n_ctx=16, ctx_init=""):
        self.n_ctx = n_ctx
        # 可学习的上下文 tokens (维度: n_ctx × 512)
        self.ctx = nn.Parameter(torch.randn(n_ctx, 512))
        
    def forward(self, image_features, class_names):
        # 生成类别文本描述
        text_descriptions = [f"a photo of a {name}" for name in class_names]
        
        # 对文本进行 token 化
        text_tokens = clip_tokenize(text_descriptions)
        
        # 在 tokens 序列中插入可学习的上下文
        # 位置: [SOS] [V1]...[Vn] [原始 tokens] [EOS]
        ctx_tokens = self.ctx.expand(text_tokens.shape[0], -1, -1)
        modified_tokens = torch.cat([ctx_tokens, text_tokens], dim=1)
        
        # 获取文本嵌入
        text_features = self.clip.encode_text(modified_tokens)
        
        # 计算相似度
        logits = image_features @ text_features.T
        return logits
```

#### 2.2.2 动态提示词优化算法 (创新点)

```python
class DynamicPromptOptimizer:
    """
    动态提示词优化器
    - 根据误分类样本自适应调整提示词
    - 困难样本加权机制
    """
    def __init__(self, n_ctx=16, alpha=0.1, beta=0.01):
        self.ctx = nn.Parameter(torch.randn(n_ctx, 512))  # 可学习 tokens
        self.alpha = alpha    # 学习率
        self.beta = beta      # 正则化系数
        self.breed_weights = {}  # 品种特定权重
        
    def compute_difficulty_weight(self, features, labels):
        """
        计算样本难度权重
        - 特征空间中远离中心的样本权重更高
        - 被频繁误分类的样本权重更高
        """
        # 计算每个品种的原型
        prototypes = {}
        for i, label in enumerate(labels):
            if label not in prototypes:
                prototypes[label] = []
            prototypes[label].append(features[i])
        
        # 计算品种中心
        for breed, feats in prototypes.items():
            center = torch.mean(torch.stack(feats), dim=0)
            self.breed_prototypes[breed] = center
            
        # 计算难度分数: 距离中心的距离
        weights = {}
        for i, (feat, label) in enumerate(zip(features, labels)):
            distance = torch.norm(feat - self.breed_prototypes[label])
            weights[i] = 1 + distance.item()
            
        return weights
    
    def adaptive_update(self, features, labels, logits):
        """
        自适应更新提示词
        - 针对困难样本增强提示词的判别能力
        """
        # 计算难度权重
        weights = self.compute_difficulty_weight(features, labels)
        
        # 计算梯度
        loss = F.cross_entropy(logits, labels, reduction='none')
        
        # 加权损失
        weighted_loss = sum(loss[i] * weights[i] for i in range(len(labels)))
        
        # 反向传播更新
        weighted_loss.backward()
        
        # 应用梯度裁剪和正则化
        with torch.no_grad():
            self.ctx.grad *= self.beta  # 正则化
            self.ctx -= self.alpha * self.ctx.grad
            
        return weighted_loss.item()
```

#### 2.2.3 细粒度品种语义增强

```python
class BreedSemanticEnhancer:
    """
    品种语义增强模块
    - 利用品种描述文本增强提示词
    - 提取品种的细粒度特征（毛长、体型、脸型等）
    """
    def __init__(self):
        # 品种特征描述模板
        self.breed_templates = {
            "cat": [
                "a {breed} cat with {coat} coat and {face} face",
                "a photo of a {breed} cat, known for its {trait}",
            ],
            "dog": [
                "a {breed} dog with {size} body and {ear} ears",
                "a photo of a {breed}, characterized by {feature}",
            ]
        }
        
        # 品种属性数据库
        self.breed_attributes = {
            "Persian": {"coat": "long fluffy", "face": "flat", "trait": "gentle temperament"},
            "Siamese": {"coat": "short", "face": "triangular", "trait": " vocal"},
            "Golden Retriever": {"size": "large", "ear": "floppy", "feature": "friendly nature"},
            # ... 更多品种
        }
    
    def get_enhanced_prompt(self, breed_name):
        """
        生成增强的提示词
        """
        attributes = self.breed_attributes.get(breed_name, {})
        templates = self.breed_templates.get("cat" if "cat" in breed_name.lower() else "dog")
        
        prompts = []
        for template in templates:
            prompt = template.format(breed=breed_name, **attributes)
            prompts.append(prompt)
            
        return prompts
```

### 2.3 可视化界面设计

```python
"""
细粒度猫狗分类可视化系统
技术栈: Streamlit + CLIP + PyTorch
"""

import streamlit as st
import torch
import clip
from PIL import Image
import matplotlib.pyplot as plt

class PetClassifierApp:
    def __init__(self):
        self.device = "cuda" if torch.cuda.is_available() else "mps"
        self.model, self.preprocess = clip.load("RN50", device=self.device)
        self.optimizer = DynamicPromptOptimizer(n_ctx=16)
        
    def load_image(self, uploaded_file):
        """加载上传的图片"""
        image = Image.open(uploaded_file).convert('RGB')
        return image
    
    def classify(self, image, breed_categories):
        """细粒度分类"""
        # 预处理
        image_input = self.preprocess(image).unsqueeze(0).to(self.device)
        
        # 图像特征
        with torch.no_grad():
            image_features = self.model.encode_image(image_input)
            image_features /= image_features.norm(dim=-1, keepdim=True)
        
        # 文本特征 (带动态提示词)
        texts = self.optimizer.generate_prompts(breed_categories)
        text_tokens = clip.tokenize(texts).to(self.device)
        
        with torch.no_grad():
            text_features = self.model.encode_text(text_tokens)
            text_features /= text_features.norm(dim=-1, keepdim=True)
        
        # 计算相似度
        logits = (image_features @ text_features.T).softmax(dim=-1)
        
        return logits, texts
    
    def visualize_attention(self, image, predictions):
        """可视化注意力区域"""
        # 生成热力图
        heatmap = self.generate_heatmap(image, predictions)
        return heatmap
    
    def render(self):
        """渲染 Streamlit 界面"""
        st.title("🐱🐕 细粒度猫狗品种分类系统")
        
        # 1. 图片上传
        uploaded_file = st.file_uploader("上传宠物图片", type=['jpg', 'png', 'jpeg'])
        
        if uploaded_file:
            image = self.load_image(uploaded_file)
            st.image(image, caption="上传图片", use_container_width=True)
            
            # 2. 分类按钮
            if st.button("开始识别"):
                # 加载品种类别
                breed_categories = self.load_breed_categories()
                
                # 执行分类
                logits, prompts = self.classify(image, breed_categories)
                
                # 显示结果
                self.show_results(logits, prompts, breed_categories)
    
    def show_results(self, logits, prompts, categories):
        """展示分类结果"""
        # Top-5 预测
        top5_probs, top5_idx = logits.topk(5)
        
        st.subheader("识别结果")
        
        col1, col2 = st.columns(2)
        
        with col1:
            st.write("**预测品种 Top-5:**")
            for i in range(5):
                breed = categories[top5_idx[0][i].item()]
                prob = top5_probs[0][i].item()
                st.progress(prob)
                st.write(f"{i+1}. {breed}: {prob:.2%}")
        
        with col2:
            st.write("**置信度热力图**")
            heatmap = self.visualize_attention(None, logits)
            st.image(heatmap, caption="关注区域", use_container_width=True)
        
        # 提示词可视化
        st.subheader("使用的提示词")
        st.code("\n".join(prompts))
```
